    note "Capa 1: Motor Genérico y Reglas de Juego (GoF Strategy & Memento)"

    class GameStatus {
        <<Enum / Value Object>>
        PLAYING
        WON
        LOST
    }

    class BaseGameState~TCell~ {
        <<interface, Value Object>>
        +board: HexBoard~TCell~
        +score: number
        +moveCount: number
        +status: GameStatus
    }

    class GameEvent {
        <<Domain Event>>
        +type: string
        +payload: Record
    }

    class ResolutionResult~TState~ {
        <<Value Object>>
        +nextState: TState
        +events: GameEvent[]
    }

    class IGameRules~TCell, TMove, TState~ {
        <<interface, Strategy>>
        +isValidMove(state: TState, move: TMove) boolean
        +applyMove(state: TState, move: TMove) TState
        +resolve(state: TState) ResolutionResult~TState~
        +evaluateStatus(state: TState) GameStatus
        +hasValidMoves(state: TState) boolean
    }

    class IScoringPolicy~TState~ {
        <<interface, Strategy>>
        +calculateTurnScore(context: TurnScoreContext) ScoreResult
    }

    class ScoreResult {
        <<Value Object>>
        +pointsEarned: number
        +newTotalScore: number
        +multiplier: number
        +starsEarned: number
        +breakdown: Record
    }

    class GameEngine~TCell, TMove, TState~ {
        <<Aggregate Root>>
        -rules: IGameRules~TCell, TMove, TState~
        -scoring: IScoringPolicy~TState~
        -pieceSource: IPieceSource~TPiece~
        -state: TState
        -history: MoveHistory~TMove, TState~
        +start(initial: TState) void
        +applyMove(move: TMove) MoveOutcome
        +undo() MoveOutcome
        +redo() MoveOutcome
    }

    class IPieceSource~TPiece~ {
        <<interface, Strategy>>
        +hasNext() boolean
        +next() TPiece
    }

    class RandomPieceSource~TPiece~ {
        -candidates: readonly TPiece[]
        -random: IRandom
        +hasNext() boolean
        +next() TPiece
    }

    class PredefinedPieceSource~TPiece~ {
        -pieces: readonly TPiece[]
        -currentIndex: number
        +hasNext() boolean
        +next() TPiece
        +remainingCount number
    }

    class IRandom {
        <<interface, Port>>
        +nextDouble() number
        +nextIndex(bound: number) number
        +nextInt(min: number, max: number) number
        +snapshot() number
        +restore(state: number) void
    }

    class SeededRandom {
        <<Domain Service, Mulberry32>>
        -state: number
        +nextDouble() number
        +nextIndex(bound: number) number
        +nextInt(min: number, max: number) number
        +snapshot() number
        +restore(state: number) void
    }

    class takeBatch {
        <<Pure Domain Utility>>
        +takeBatch(source, count) readonly TPiece[]
    }

    class MoveHistory~TMove, TState~ {
        <<Value Object, Memento>>
        -undoStack: MoveCommand[]
        -redoStack: MoveCommand[]
        +push(command) void
        +undo() MoveCommand
        +redo() MoveCommand
    }

    BaseGameState o-- GameStatus : contiene
    ResolutionResult o-- GameEvent : contiene
    IGameRules ..> ResolutionResult : produce en resolve
    IScoringPolicy ..> ScoreResult : produce
    GameEngine o-- BaseGameState : gestiona
    GameEngine o-- IGameRules : estrategia reglas
    GameEngine o-- IScoringPolicy : estrategia puntaje
    GameEngine o-- IPieceSource : generador piezas
    GameEngine o-- MoveHistory : historial undo/redo
    RandomPieceSource ..|> IPieceSource : implementa
    PredefinedPieceSource ..|> IPieceSource : implementa
    RandomPieceSource o-- IRandom : consume
    SeededRandom ..|> IRandom : implementa
    takeBatch ..> IPieceSource : consume lote (SRP/ISP)
